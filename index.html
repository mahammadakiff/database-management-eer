Subclass PK = Superclass PK + FK to the superclass.One relation per class (Elmasri & Navathe mapping option 8A): each subclass table reuses char_id as its primary key and as a foreign key to CHARACTER.

CREATE TABLE

-- One relation per class (mapping option 8A).
-- CHARACTER is a reserved word in SQL, so the name is quoted
-- (standard SQL double quotes; use backticks in MySQL).
CREATE TABLE "CHARACTER" (
  char_id   INT          PRIMARY KEY,
  name      VARCHAR(50)  NOT NULL,
  level     INT          NOT NULL
);

CREATE TABLE WARRIOR (
  char_id   INT          PRIMARY KEY,
  strength  INT          NOT NULL,
  FOREIGN KEY (char_id) REFERENCES "CHARACTER"(char_id)
    ON DELETE CASCADE
);

CREATE TABLE MAGE (
  char_id   INT          PRIMARY KEY,
  mana      INT          NOT NULL,
  FOREIGN KEY (char_id) REFERENCES "CHARACTER"(char_id)
    ON DELETE CASCADE
);

CREATE TABLE ROGUE (
  char_id   INT          PRIMARY KEY,
  stealth   INT          NOT NULL,
  FOREIGN KEY (char_id) REFERENCES "CHARACTER"(char_id)
    ON DELETE CASCADE
);

-- Aria (char_id 1) has a row in WARRIOR and in MAGE (overlapping).
